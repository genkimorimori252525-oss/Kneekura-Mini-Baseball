  const z=data.strikeZoneBatterPov;
  if(z.every(v=>v!==null)){
    mc.strokeStyle="#f5c542";mc.lineWidth=1;
    mc.beginPath();mc.moveTo(z[0],z[1]);mc.lineTo(z[2],z[3]);mc.lineTo(z[6],z[7]);mc.lineTo(z[4],z[5]);mc.closePath();mc.stroke();

function loop(now){
  if(playing){
    const dt=Math.min(50,now-lastTime);lastTime=now;currentUs+=dt*1000;
    if(currentUs>=currentCourse().durationMicros){currentUs=currentCourse().durationMicros;playing=false;}
    render();
  }
  requestAnimationFrame(loop);
}
